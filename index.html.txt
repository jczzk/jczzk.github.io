<script>
    function toggleSecoes(opcao) {
        const secaoSim = document.getElementById('secaoSim');
        const secaoNao = document.getElementById('secaoNao');
        
        if (opcao === 'sim') {
            secaoSim.style.display = 'block';
            secaoNao.style.display = 'none';
        } else {
            secaoSim.style.display = 'none';
            secaoNao.style.display = 'block';
        }
    }

    document.getElementById('projetoForm').addEventListener('submit', function(e) {
        e.preventDefault();

        const nome = document.getElementById('nome').value;
        const ra = document.getElementById('ra').value;
        const whatsapp = document.getElementById('whatsapp').value;
        const turno = document.getElementById('turno_aula').value;
        const participar = document.querySelector('input[name="participar"]:checked')?.value;

        let texto = `*FICHA - PROJETO ADS (2º SEMESTRE)*\n`;
        texto += `👤 *Nome:* ${nome} (RA: ${ra})\n`;
        texto += `📱 *WhatsApp:* ${whatsapp} | *Turno:* ${turno}\n`;

        if (participar === 'sim') {
            const frentes = Array.from(document.querySelectorAll('input[name="frente"]:checked')).map(el => el.value);
            const nivel = document.getElementById('nivel_exp').value;
            const equip = document.getElementById('equipamento').value;
            const tech = document.getElementById('tecnologia').value || 'A definir na reunião';
            const objetivo = document.querySelector('input[name="objetivo"]:checked')?.value || 'Não informado';
            const lider = document.querySelector('input[name="lideranca"]:checked')?.value || 'Não informado';
            const horas = document.querySelector('input[name="horas_semanais"]:checked')?.value || 'Não informado';
            const formato = document.querySelector('input[name="formato_reuniao"]:checked')?.value || 'Não informado';
            const horarios = Array.from(document.querySelectorAll('input[name="horario_grade"]:checked')).map(el => el.value);
            const obs = document.getElementById('obs_horario').value;

            if (frentes.length === 0) {
                alert('Por favor, selecione pelo menos uma frente de atuação.');
                return;
            }

            texto += `✅ *Status:* VAI PARTICIPAR\n`;
            texto += `💻 *Frentes:* ${frentes.join(', ')}\n`;
            texto += `📊 *Nível:* ${nivel} | *PC:* ${equip}\n`;
            texto += `🛠️ *Sugestão Stack:* ${tech}\n`;
            texto += `🎯 *Objetivo:* ${objetivo} | *Liderança:* ${lider}\n`;
            texto += `⏱️ *Dedicação:* ${horas} | *Reuniões:* ${formato}\n`;
            texto += `📅 *Horários Livres:* ${horarios.length > 0 ? horarios.join(', ') : 'Nenhum marcado'}\n`;
            if (obs) texto += `📝 *Obs:* ${obs}\n`;

        } else {
            const motivo = document.querySelector('input[name="motivo"]:checked')?.value;
            const acompanhar = document.querySelector('input[name="acompanhar"]:checked')?.value;

            if (!motivo || !acompanhar) {
                alert('Por favor, responda o motivo e se deseja acompanhar o grupo.');
                return;
            }

            texto += `❌ *Status:* NÃO VAI PARTICIPAR\n`;
            texto += `📌 *Motivo:* ${motivo}\n`;
            texto += `🔔 *Continuar no grupo:* ${acompanhar}\n`;
        }

        document.getElementById('resumoTexto').value = texto;
        document.getElementById('resumoBox').style.display = 'block';
        document.getElementById('resumoBox').scrollIntoView({ behavior: 'smooth' });
    });

    function copiarResumo() {
        const copyText = document.getElementById('resumoTexto');
        copyText.select();
        copyText.setSelectionRange(0, 99999);
        navigator.clipboard.writeText(copyText.value);
        alert('Resumo copiado! Agora é só colar no WhatsApp.');
    }
</script>
